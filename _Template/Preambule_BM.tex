\makeatletter

% Récupère le numéro de page de chaque section dans la TOC
\patchcmd{\beamer@sectionintoc}
{\def\inserttocsection{\hyperlink{Navigation#3}{#2}}}
{%
  \def\inserttocsection{\hyperlink{Navigation#3}{#2}}%
  \def\inserttocsectionpage{#3}%
}
{}{}

% Récupère le numéro de page de chaque sous-section dans la TOC
\patchcmd{\beamer@subsectionintoc}
{\def\inserttocsubsection{\hyperlink{Navigation#4}{#3}}}
{%
  \def\inserttocsubsection{\hyperlink{Navigation#4}{#3}}%
  \def\inserttocsubsectionpage{#4}%
}
{}{}

\makeatother

% Section
\setbeamertemplate{section in toc}{%
  \leavevmode
  \leftskip=0em
  \rightskip=0em
  \parfillskip=0pt
  \usebeamerfont{section in toc}%
  \textcolor{couleurmatiere}{\inserttocsectionnumber.}\hspace{0.5em}%
  \inserttocsection
  \nobreak\leaders\hbox to 0.6em{\hss.\hss}\hfill
  \inserttocsectionpage
  \par
}

% Sous-section
\setbeamertemplate{subsection in toc}{%
  \leavevmode
  \leftskip=2em
  \rightskip=0em
  \parfillskip=0pt
  \usebeamerfont{subsection in toc}%
  \textcolor{couleurmatiere}{\inserttocsectionnumber.\inserttocsubsectionnumber}%
  \hspace{0.5em}%
  \inserttocsubsection
  \nobreak\leaders\hbox to 0.6em{\hss.\hss}\hfill
  \inserttocsubsectionpage
  \par
}

% On cache les sous-sous-sections
\setbeamertemplate{subsubsection in toc}{}

% Apparence générale
\setbeamerfont{section in toc}{size=\large}
\setbeamerfont{subsection in toc}{size=\normalsize}
